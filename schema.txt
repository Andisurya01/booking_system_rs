// ─────────────────────────────────────────────────────────────────────────────
// Safe types (tanpa field sensitif)
// ─────────────────────────────────────────────────────────────────────────────

export type SafeUser = Omit<User, 'password'>

export type SafePatient = Omit<Patient, 'nik' | 'url_image_nik'>

// ─────────────────────────────────────────────────────────────────────────────
// With relations (untuk response yang include relasi)
// ─────────────────────────────────────────────────────────────────────────────

export interface UserWithRoles extends SafeUser {
  roles: (UserRole & { role: Role })[]
}

export interface PatientWithInsurances extends Patient {
  insurances: (PatientInsurance & { insurance: Insurance })[]
}

export interface DoctorWithSpecializations extends Doctor {
  user: SafeUser
  specializations: (DoctorSpecialization & { specialization: Specialization })[]
}

export interface DoctorScheduleWithRelations extends DoctorSchedule {
  doctor: Doctor & { user: Pick<User, 'id' | 'name'> }
  room: Room | null
}

export interface RegistrationWithRelations extends Registration {
  patient: Pick<Patient, 'id' | 'uuid' | 'name' | 'medical_record_number'>
  schedule: DoctorScheduleWithRelations
  insurance: PatientInsurance | null
  user: Pick<User, 'id' | 'name'>
}
